5. Sequence Diagram - Live Quiz Game Flow

%%{init: {'theme': 'neutral'}}%%
sequenceDiagram
    autonumber
    
    actor Student as Student
    participant Browser as Browser
    participant WebSocket as WebSocket Server
    participant DB as Database
    actor Host as Host/Teacher

    rect rgb(240, 248, 255)
        Note over Host,DB: Game Setup Phase
        Host->>WebSocket: Create game room
        WebSocket->>DB: Store room (room_code, quiz_id)
        WebSocket-->>Host: Room code generated
    end

    rect rgb(255, 248, 240)
        Note over Student,DB: Join Phase
        Student->>Browser: Enter room code
        Browser->>WebSocket: emit("join_game", room_code)
        WebSocket->>DB: Verify room exists
        
        alt Room Exists
            DB-->>WebSocket: Room details
            WebSocket->>DB: Add participant
            WebSocket-->>Browser: emit("player_joined")
            WebSocket-->>Host: emit("player_list_updated")
        else Room Not Found
            DB-->>WebSocket: Error (Not Found)
            WebSocket-->>Browser: emit("error", "Invalid Room")
            Browser-->>Student: Show Error Message
        end
    end

    rect rgb(240, 255, 240)
        Note over Host,DB: Game Start
        Host->>WebSocket: emit("start_game")
        WebSocket-->>Browser: emit("game_started")
    end

    loop For Each Question
        rect rgb(255, 255, 240)
            Note over Student,DB: Question Round
            WebSocket->>Browser: emit("new_question", data)
            Browser->>Student: Display question + timer
            Student->>Browser: Select answer
            Browser->>WebSocket: emit("submit_answer", answer)
            WebSocket->>WebSocket: Calculate score (time bonus)
            WebSocket->>DB: Save response
            WebSocket-->>Browser: emit("answer_result")
            WebSocket-->>Host: emit("leaderboard_update")
        end
    end

    rect rgb(240, 240, 255)
        Note over Student,DB: Game End
        WebSocket->>DB: Update XP, save final scores
        WebSocket-->>Browser: emit("game_ended", results)
        WebSocket-->>Host: emit("final_results")
    end

7. State Diagram - Game Room Lifecycle

%%{init: {'theme': 'neutral'}}%%
stateDiagram-v2
    [*] --> Created: Host creates room
    
    Created --> Waiting: Room code generated
    
    Waiting --> Active: Host starts game (min 1 player)
    Waiting --> Cancelled: Host cancels or timeout
    
    state Active {
        [*] --> QuestionPhase
        QuestionPhase --> AnswerCollection: Timer starts
        AnswerCollection --> Scoring: Timer ends or all answered
        Scoring --> LeaderboardUpdate: Calculate scores
        LeaderboardUpdate --> QuestionPhase: Next question
        LeaderboardUpdate --> [*]: No more questions
    }
    
    Active --> GameComplete: All questions done
    GameComplete --> Finished: Save results, update XP
    Cancelled --> [*]
    Finished --> [*]
